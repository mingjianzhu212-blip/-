const doBump = () => {
  load();                       /* 锁内重读，以磁盘为准 */
  ensureDay();
  const before = st.prog[kind]; /* 写盘失败要回滚到这个"修改前"的进度 */
  st.prog[kind] = Math.min(t.goal, (before || 0) + add);
  /* save() 返回布尔。写盘失败时内存里的 +add 没落盘，必须回滚，
     否则 list() 会把没保存的进度当真进度显示 —— 那就是"假装已经记下"，
     与本模块对写盘失败的一贯口径（doClaim 早就回滚 + 返回失败）相冲突。 */
  if(!save()){
    st.prog[kind] = before;
    return { ok:false, code:'SAVE_FAILED', msg:'任务进度保存失败（本机存储异常），本次进度未记下。' };

  return { ok:true, prog: st.prog[kind] };
};

**③ dailyBump() 对普通保存失败提示、与无锁区分、不刷屏**（js/game.js:4303-4318）：

let dailyNoLockWarned = false;
let dailySaveFailWarned = false;
function dailyBump(kind){
  if(typeof Daily === 'undefined') return;
  Promise.resolve(Daily.bump(kind)).then(r => {
    if(r && r.ok === false && r.code === 'NO_LOCK' && !dailyNoLockWarned){
      dailyNoLockWarned = true;
      log('⚠️ ' + r.msg + '（日常任务进度本次不记，其他玩法不受影响）', 'bad');
    } else if(r && r.ok === false &&
             (r.code === 'SAVE_FAILED' || r.code === 'BUMP_FAILED') &&
             !dailySaveFailWarned){
      /* 只认这两个码；UNKNOWN_TASK（herb/enlight/ponder 等正常动作）静默忽略，
         否则每次采药/顿悟都误报"保存失败"。 */
      dailySaveFailWarned = true;
      log('⚠️ ' + r.msg + '（可清理本机存储后重试，其他玩法不受影响）', 'bad');
    }
  }, () => {});
}

- 两个独立 warned 标志：无锁提示、写盘失败提示各只一次，互不抢占，不刷屏。
