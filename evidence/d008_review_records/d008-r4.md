## 一、第三轮遗留项（唯一一项）

> 问题不在 js/game.js，而在新增的类型校验器：validBoost、validRoll 本身能识别非法值，但调用它们的条件仍使用了真值判断，导致部分非法的嵌套值根本没有进入校验。
> 例：当前 verify.js `if(fx.boost){ if(!validBoost(fx.boost)) badType++; }`——若 `boost: null`，`if(fx.boost)` 为假，validBoost 不执行，漏检。
> 修复：把这最后一个真值判断漏洞补上，即具备签发「审查通过 · 可交付」的条件。

## 二、定点修复（verify.js 与 test 一致）

### 2.1 verify.js · chkFx 守卫由真值判断改为 `!== undefined`
```js
/* 必须用不等于 undefined 判断，而非真值判断；否则 boost:null 或 roll:null 等 falsy 非法值
   会被 if(fx.boost) 直接跳过，绕过 validBoost/validRoll（第三轮审查补证点）。 */
if(fx.boost !== undefined){
  const bk = Object.keys(fx.boost || {});
  if(!validBoost(fx.boost)) badType++;
  else if(bk.some(k => k !== 'mult' && k !== 'sec')) badShape++;
  else if(!inSet(ALLOW.mult, fx.boost.mult) || !inSet(ALLOW.sec, fx.boost.sec)) badTier++;
}
if(fx.roll !== undefined){
  if(!validRoll(fx.roll)) badType++;
  else fx.roll.forEach(b => {
    if(Object.keys(b).some(k => k !== 'w' && k !== 'text' && k !== 'fx')) badShape++;
    if(typeof b.w === 'number' && !inSet(ALLOW.w, b.w)) badTier++;
    chkFx(b.fx);
  });
}
```
`validBoost(null)` → `null !== null`（false）且 `typeof null === 'object'` 但 `Array.isArray` 否、`.every` 不成立 → 返回 false → `badType++`。`validRoll(null)` → `Array.isArray(null)` 否 → false → `badType++`。即 null 进入校验后被正确判为坏类型。

### 2.2 tools/d008-content.test.mjs · 同步守卫 + 新增漏检反例
结构测试 chk 同步改为：
```js
const chk = (fx, id) => {
  if(!validFx(fx)) return;
  Object.keys(fx).forEach(k => assert.ok(FX_KEYS.indexOf(k) >= 0, id + ' fx 键 ' + k + ' 合法'));
  if(fx.boost !== undefined) assert.ok(validBoost(fx.boost), id + ' boost 为合法对象且仅含 mult/sec');
  if(fx.roll !== undefined){
    assert.ok(Array.isArray(fx.roll), id + ' roll 为数组');
    fx.roll.forEach(b => {
      assert.strictEqual(typeof b.w, 'number', id + ' roll.w 是数字');
      assert.ok(b.text && typeof b.text === 'string', id + ' roll.text 存在');
      assert.ok(validFx(b.fx), id + ' roll.fx 为合法对象');
      chk(b.fx, id);
    });
  }
};
```
新增专项反例（证明漏检已闭合，模拟与 verify.js 同款守卫逻辑）：
```js
test('fx 类型契约：嵌套 falsy 非法值（boost:null / roll:null）漏检漏洞已闭合', () => {
  const safeChk = (fx) => {
    if(!validFx(fx)) return false;
    if(fx.boost !== undefined && !validBoost(fx.boost)) return false;
    if(fx.roll !== undefined && !validRoll(fx.roll)) return false;
    return true;
  };
  assert.strictEqual(validBoost(null), false, 'boost:null 被拒');
  assert.strictEqual(validBoost(0), false, 'boost:0 被拒');
  assert.strictEqual(validRoll(null), false, 'roll:null 被拒');
  assert.strictEqual(validRoll('x'), false, 'roll:字符串 被拒');
  assert.strictEqual(safeChk({ boost: null }), false, '{boost:null} 被拒（不再漏检）');
  assert.strictEqual(safeChk({ roll: null }), false, '{roll:null} 被拒（不再漏检）');
  assert.strictEqual(safeChk({ boost: { mult: 0.6, sec: 900 } }), true, '合法 boost 通过');
  assert.strictEqual(safeChk({ roll: [{ w: 6, text: 'a', fx: {} }] }), true, '合法 roll 通过');
});
```
