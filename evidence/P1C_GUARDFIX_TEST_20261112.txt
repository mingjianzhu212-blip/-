function confirmCharWrite(charPath, SPath){
  // 复刻真实守卫：写入前校验 incoming charPath 属 PATH_IDS，且本源尚未确定
  let S = { path: SPath };
  if(PATH_IDS.indexOf(charPath) >= 0 && (typeof S.path !== 'string' || PATH_IDS.indexOf(S.path) < 0)) S.path = charPath;
  return S.path;
}
// A 未选本源 + 合法 charPath -> 写入
